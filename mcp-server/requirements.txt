# `from tools.crud import ...`, i.e. it treats src/ as the import root. Those
# imports only resolve once the PROJECT ITSELF is installed. With dependencies
# alone you get 13 collection errors of the form:
#
#     ModuleNotFoundError: No module named 'tools'
#

#
#     python3 -m venv .venv
#     .venv/bin/pip install -r requirements.txt
#     .venv/bin/pip install -e .          # <- required; installs src/ as a package
#     .venv/bin/python -m pytest tests/ -q
#
